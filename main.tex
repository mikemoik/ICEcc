% Custom function: \drawredcross{x}{y}{scale}
\newcommand{\drawredcross}[2]{
    \begin{scope}[shift={#1}, scale=#2]
        \fill[redcrossred] 
            (-1.5, -0.5) rectangle (1.5, 0.5)  % Horizontal bar
            (-0.5, -1.5) rectangle (0.5, 1.5); % Vertical bar
    \end{scope}
}
