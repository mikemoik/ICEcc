% Custom function: \drawredcross{x}{y}{scale}
\newcommand{\drawredcross}[2]{
  \begin{scope}[shift={#1}, scale=#2]
    \fill[redcrossred] 
        (-1.5, 0.5) rectangle (1.5, -0.5)  % Horizontal bar
        (-0.5, 1.5) rectangle (0.5, -1.5); % Vertical bar
  \end{scope}
}
\newcommand{\drawredmoon}[2]{
  \begin{scope}[shift={#1}, scale=#2]
    \clip (-1.5,-1.5) rectangle (1.5,1.5); % Ensure canvas size
    \fill[redcrossred] (0,0) circle (1cm); % Outer circle
    \fill[white] (0.35,0) circle (0.8cm);  % Inner circle (offset to the right)
  \end{scope}
}
\newcommand{\drawredcrystal}[2]{
  \begin{scope}[shift={#1}, scale=#2, rotate=45]
    \fill[redcrossred] (-1,1) rectangle (1,-1); % Outer rectangle
    \fill[white] (-.5,.5) rectangle (.5,-.5); % inner rectangle
  \end{scope}
}
